🛠️ Supabase SQL Şeması

Supabase projenizde SQL Editor alanına girerek sorgu ekranına aşağıdaki kodun tamamını yapıştırıp RUN tuşuna basın. Bu kod, gerekli tabloları, otomatik üye onay mekanizmasını ve güvenlik filtrelerini (RLS) kurar.

create extension if not exists "uuid-ossp";

-- 2. Roller Tablosu
create table if not exists public.roles (
  id uuid default uuid_generate_v4() primary key,
  name text not null unique
);

-- Default rolleri ekle
insert into public.roles (name) values ('admin'), ('member') on conflict (name) do nothing;

);

-- 10. Onay Kontrol Fonksiyonları (Sıkı Admin Güvenliği)
alter table public.roles enable row level security;

drop policy if exists "Authenticated users can read roles" on public.roles;
create policy "Authenticated users can read roles" on public.roles
  for select using (auth.uid() is not null);

create or replace function public.is_approved_user(user_uuid uuid)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
begin
  if user_uuid is null then
    return false;
  end if;
  return exists (
    select 1 from public.profiles
    where id = user_uuid and status = 'approved'
  );
end;
$$;

create or replace function public.is_admin_user(user_uuid uuid)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
begin
  if user_uuid is null then
    return false;
  end if;
  return exists (
    select 1 from public.profiles p
    join public.roles r on p.role_id = r.id
    where p.id = user_uuid
      and p.status = 'approved'
      and r.name = 'admin'
  );
end;
$$;

-- 11. Profil Güvenlik Politikaları
drop policy if exists "Users can read their own profile" on public.profiles;
create policy "Users can read their own profile" on public.profiles
  for select using (auth.uid() = id);

drop policy if exists "Users can insert their own profile" on public.profiles;
create policy "Users can insert their own profile" on public.profiles
  for insert with check (
    auth.uid() = id
    and status = 'pending'
    and (role_id is null or role_id <> (select id from public.roles where name = 'admin' limit 1))
  );

drop policy if exists "Admins can perform all actions on all profiles" on public.profiles;
create policy "Admins can perform all actions on all profiles" on public.profiles
  for all using (public.is_admin_user(auth.uid()));

-- 12. Yeni Kayıtlarda Otomatik Profil Oluşturma Tetikleyicisi
create or replace function public.handle_new_user()
returns trigger as $$
declare
  is_first bool;
  is_admin_email bool;
  role_id_var uuid;
begin
  select not exists (select 1 from public.profiles) into is_first;
  is_admin_email := (new.email = 'bllacrr@gmail.com');
  
  -- Rol belirleme
  if (is_first or is_admin_email) then
    select id into role_id_var from public.roles where name = 'admin' limit 1;
  else
    select id into role_id_var from public.roles where name = 'member' limit 1;
  end if;
  
  insert into public.profiles (id, email, status, role_id)
  values (
    new.id, 
    new.email, 
    case when (is_first or is_admin_email) then 'approved' else 'pending' end, 
    role_id_var
  );
  return new;
end;
$$ language plpgsql security definer;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- 13. Mevcut Kullanıcıları Profile Senkronize Etme ve İlkini Admin Yapma
insert into public.profiles (id, email, status, role_id)
select id, email, 'approved', (select id from public.roles where name = 'admin' limit 1) 
from auth.users
on conflict (id) do nothing;

update public.profiles
set status = 'approved',
    role_id = (select id from public.roles where name = 'admin' limit 1)
where id in (select id from public.profiles order by created_at asc limit 1)
   or email = 'bllacrr@gmail.com';

  p_page_path text default '#explore',
  p_post_slug text default null,
  p_post_title text default null,
  p_referrer text default null,
  p_referrer_domain text default null,
  p_utm_source text default null,
  p_utm_medium text default null,
  p_utm_campaign text default null,
  p_device_type text default null,
  p_os text default null,
  p_browser text default null,
  p_language text default null,
  p_screen_resolution text default null,
  p_timezone text default null,
  p_country text default null,
  p_city text default null,
  p_region text default null,
  p_is_pwa boolean default false,
  p_user_id uuid default null,
  p_user_email text default null,
  p_scroll_depth integer default null,
  p_duration_seconds integer default null,
  p_search_query text default null,
  p_search_results_count integer default null
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_auth_uid uuid := auth.uid();
  v_safe_email text := null;
  v_is_auth_click boolean := (p_event_type = 'auth_cta_click' or coalesce(p_page_path, '') like '#auth%');
  v_safe_scroll integer := case when p_scroll_depth is not null then least(greatest(p_scroll_depth, 0), 100) else null end;
  v_safe_dur integer := case when p_duration_seconds is not null then least(greatest(p_duration_seconds, 0), 3600) else null end;
begin
  if p_visitor_id is null or length(trim(p_visitor_id)) < 6 then
    return;
  end if;

  if v_auth_uid is not null then
    select email into v_safe_email from public.profiles where id = v_auth_uid limit 1;
    v_safe_email := coalesce(v_safe_email, p_user_email);
  end if;

  insert into public.site_visitors (
    visitor_id, user_id, user_email, first_seen_at, last_seen_at, visit_count,
    first_referrer, first_referrer_domain, first_landing_page, last_page,
    first_utm_source, first_utm_medium, first_utm_campaign,
    device_type, os, browser, language, screen_resolution, timezone,
    country, city, region, is_pwa, clicked_auth, max_scroll_depth, total_read_seconds
  ) values (
    p_visitor_id, v_auth_uid, v_safe_email, timezone('utc'::text, now()), timezone('utc'::text, now()), 1,
    p_referrer, p_referrer_domain, p_page_path, p_page_path,
    p_utm_source, p_utm_medium, p_utm_campaign,
    p_device_type, p_os, p_browser, p_language, p_screen_resolution, p_timezone,
    p_country, p_city, p_region, coalesce(p_is_pwa, false),
    v_is_auth_click, coalesce(v_safe_scroll, 0), coalesce(v_safe_dur, 0)
  )
  on conflict (visitor_id) do update set
    last_seen_at = timezone('utc'::text, now()),
    visit_count = public.site_visitors.visit_count + case when p_event_type in ('page_view', 'post_view') then 1 else 0 end,
    last_page = coalesce(p_page_path, public.site_visitors.last_page),
    user_id = coalesce(v_auth_uid, public.site_visitors.user_id),
    user_email = coalesce(v_safe_email, public.site_visitors.user_email),
    device_type = coalesce(p_device_type, public.site_visitors.device_type),
    os = coalesce(p_os, public.site_visitors.os),
    browser = coalesce(p_browser, public.site_visitors.browser),
    country = coalesce(p_country, public.site_visitors.country),
    city = coalesce(p_city, public.site_visitors.city),
    region = coalesce(p_region, public.site_visitors.region),
    is_pwa = coalesce(p_is_pwa, public.site_visitors.is_pwa),
    clicked_auth = (public.site_visitors.clicked_auth or v_is_auth_click),
    max_scroll_depth = greatest(coalesce(public.site_visitors.max_scroll_depth, 0), coalesce(v_safe_scroll, 0)),
    total_read_seconds = coalesce(public.site_visitors.total_read_seconds, 0) + coalesce(v_safe_dur, 0);

  insert into public.site_events (
    visitor_id, session_id, user_id, event_type, page_path, post_slug, post_title,
    referrer_domain, country, city, device_type,
    scroll_depth, duration_seconds, search_query, search_results_count
  ) values (
    p_visitor_id, p_session_id, v_auth_uid, coalesce(p_event_type, 'page_view'),
    p_page_path, p_post_slug, p_post_title, p_referrer_domain, p_country, p_city, p_device_type,
    v_safe_scroll, v_safe_dur, p_search_query, p_search_results_count
  );
end;
$$;

create or replace function public.link_visitor_to_user(
  p_visitor_id text,
  p_user_id uuid,
  p_user_email text default null
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_auth_uid uuid := auth.uid();
  v_safe_email text;
begin
  if v_auth_uid is null or v_auth_uid <> p_user_id then
    raise exception 'Unauthorized visitor link attempt';
  end if;

  select email into v_safe_email from public.profiles where id = v_auth_uid limit 1;
  v_safe_email := coalesce(v_safe_email, p_user_email);

  update public.site_visitors
  set user_id = v_auth_uid,
      user_email = coalesce(v_safe_email, user_email),
      last_seen_at = timezone('utc'::text, now())
  where visitor_id = p_visitor_id;

  update public.site_events
  set user_id = v_auth_uid
  where visitor_id = p_visitor_id and user_id is null;
end;
$$;
